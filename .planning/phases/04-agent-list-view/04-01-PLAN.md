   - Get current model from `local.model.current()` - if `providerID === "agentos"`, that's the connected agent

   - Only handle when `store.activeTab === "agents"` and `store.selectedAgent === null`
   - `up` / `ctrl+p`: move selection up (wrap to end)
   - `down` / `ctrl+n`: move selection down (wrap to start)
   - `return`: set `selectedAgent` to the currently highlighted agent
   - When selectedAgent is set, `escape` should clear it (back to list)

7. **Replace the agents tab content placeholder with scrollbox list:**
   ```tsx
   <Show when={store.activeTab === "agents"}>
     <Show when={!store.selectedAgent} fallback={<AgentDetail agent={store.selectedAgent} onBack={() => setStore("selectedAgent", null)} />}>
       <Show when={filteredAgents().length > 0} fallback={<text fg={theme.textMuted}>No agents found</text>}>
         <scrollbox ref={...} maxHeight={height()} scrollbarOptions={{ visible: false }}>
           <For each={filteredAgents()}>
             {(agent, index) => (
               <AgentRow
                 agent={agent}
                 active={index() === store.selectedIndex}
                 onSelect={() => setStore("selectedAgent", agent)}
                 onHover={() => setStore("selectedIndex", index())}
               />
             )}
           </For>
         </scrollbox>
       </Show>
     </Show>
   </Show>
   ```

8. **Create AgentRow component (inline in same file):**
   ```tsx
   function AgentRow(props: {
     agent: { id: string; name: string; isConnected: boolean };
     active: boolean;
     onSelect: () => void;
     onHover: () => void;
   }) {
     const { theme } = useTheme()
     const fg = selectedForeground(theme)

     return (
       <box
         flexDirection="row"
         backgroundColor={props.active ? theme.primary : RGBA.fromInts(0, 0, 0, 0)}
         paddingLeft={props.agent.isConnected ? 1 : 3}
         paddingRight={3}
         gap={1}
         onMouseOver={props.onHover}
         onMouseUp={props.onSelect}
       >
         <Show when={props.agent.isConnected}>
           <text flexShrink={0} fg={props.active ? fg : theme.primary}>
             ●
           </text>
         </Show>
         <text
           flexGrow={1}
           fg={props.active ? fg : props.agent.isConnected ? theme.primary : theme.text}
           attributes={props.active ? TextAttributes.BOLD : undefined}
         >
           {props.agent.name}
         </text>
       </box>
     )
   }
   ```

9. **Create AgentDetail placeholder component (inline):**
   ```tsx
   function AgentDetail(props: { agent: { id: string; name: string }; onBack: () => void }) {
     const { theme } = useTheme()

     useKeyboard((evt) => {
       if (evt.name === "escape") {
         props.onBack()
         evt.preventDefault()
       }
     })

     return (
       <box flexDirection="column" gap={1}>
         <text fg={theme.text} attributes={TextAttributes.BOLD}>{props.agent.name}</text>
         <text fg={theme.textMuted}>Agent details coming in Phase 5</text>
         <text fg={theme.textMuted}>Press Esc to go back</text>
       </box>
     )
   }
   ```
